2.Queries úteis e campos

%%bash 
%%bash
# nvidia-smi — query estruturado (GPU real ou simulado)
if command -v nvidia-smi >/dev/null 2>&1; then
  nvidia-smi --query-gpu=index,name,temperature.gpu,utilization.gpu,\
utilization.memory,memory.used,memory.total,power.draw,power.limit,\
clocks.current.graphics,clocks.current.memory --format=csv,noheader,nounits
  echo "--- processos na GPU ---"
  nvidia-smi --query-compute-apps=pid,process_name,used_gpu_memory --format=csv,noheader
else
  echo "Sem GPU NVIDIA — saída simulada (Tesla T4):"
  echo "0, Tesla T4, 52, 87, 45, 3842, 15360, 48.2, 70.0, 1590, 5001"
fi

csv,noheader,nounits remove cabeçalho e unidades — perfeito para juntar com timestamp e gravar no CSV.

🌡️ Temperatura e energia

  • temperature.gpu — °C atual
  • power.draw — watts consumidos
  • power.limit — limite configurado

📈 Utilização e memória

  • utilization.gpu — % uso dos SMs
  • memory.used / memory.total — VRAM

3.monitor_gpu.sh e alerta_gpu.sh

Coleta em CSV

%%bash 
%%bash
# monitor_gpu.sh (núcleo da aula) — coleta N amostras em CSV
INTERVALO=1; AMOSTRAS=5; SAIDA="gpu_log.csv"
CABECALHO="timestamp,gpu_index,gpu_name,temp_c,util_gpu_pct,"
CABECALHO+="util_mem_pct,mem_used_mb,mem_total_mb,power_w,power_limit_w"
echo "$CABECALHO" > "$SAIDA"

for ((i=0; i<AMOSTRAS; i++)); do
  TS=$(date "+%Y-%m-%d %H:%M:%S")
  if command -v nvidia-smi >/dev/null 2>&1; then
    nvidia-smi --query-gpu=index,name,temperature.gpu,utilization.gpu,\
utilization.memory,memory.used,memory.total,power.draw,power.limit \
      --format=csv,noheader,nounits \
      | while IFS= read -r l; do echo "$TS,$l" >> "$SAIDA"; done
  else
    TEMP=$((60 + i * 6))   # 60,66,72,78,84 -> dispara alerta no fim
    echo "$TS,0,Tesla T4 (sim),$TEMP,80,50,8000,15360,60,70" >> "$SAIDA"
  fi
  sleep "$INTERVALO"
done
cat "$SAIDA"

set -euo pipefail no script real aborta em erro, variável vazia ou falha de pipe.

Alerta por limites

%%bash 
%%bash
# alerta_gpu.sh (núcleo) — checa limites de temperatura/utilização
LIMITE_TEMP=80; LIMITE_UTIL=95

if command -v nvidia-smi >/dev/null 2>&1; then
  dados=$(nvidia-smi --query-gpu=index,name,temperature.gpu,utilization.gpu --format=csv,noheader,nounits)
elif [ -f gpu_log.csv ]; then
  # Sem GPU: monta a linha a partir da última amostra do CSV gerado acima
  IFS=, read -r _ idx nome temp util _ < <(tail -n 1 gpu_log.csv)
  dados="$idx, $nome, $temp, $util"
else
  dados="0, Tesla T4 (sim), 84, 97"
fi

while IFS="," read -r idx nome temp util; do
  idx=$(echo "$idx" | tr -d ' ')
  nome=$(echo "$nome" | sed 's/^ *//; s/ *$//')
  temp=$(echo "$temp" | tr -d ' ')
  util=$(echo "$util" | tr -d ' ')
  [[ "$temp" =~ ^[0-9]+$ ]] && [ "$temp" -ge "$LIMITE_TEMP" ] \
    && echo "[$(date +%H:%M:%S)] ALERTA: GPU $idx ($nome): ${temp}C >= ${LIMITE_TEMP}C" || true
  [[ "$util" =~ ^[0-9]+$ ]] && [ "$util" -ge "$LIMITE_UTIL" ] \
    && echo "[$(date +%H:%M:%S)] ALERTA: GPU $idx ($nome): ${util}% >= ${LIMITE_UTIL}%" || true
done <<< "$dados"

Notifica por Slack (webhook) ou e-mail e registra em log. O script completo trata N/A e usa IFS="," com trim.

# Google Sheets — envia gpu_log.csv (valida mesmo sem credenciais)
import csv, os

SHEETS_ID = os.environ.get("SHEETS_ID", "")
CRED_FILE = os.environ.get("GOOGLE_CREDS", "service_account.json")

with open("gpu_log.csv", encoding="utf-8") as f:
    reader = csv.reader(f)
    cabecalho = next(reader)
    linhas = list(reader)

if not SHEETS_ID or not os.path.exists(CRED_FILE):
    print("Sem credenciais — modo validação.")
    print(f"{len(cabecalho)} colunas, {len(linhas)} linhas prontas para envio.")
    print("Cabeçalho:", cabecalho)
else:
    from google.oauth2 import service_account
    from googleapiclient.discovery import build
    creds = service_account.Credentials.from_service_account_file(
        CRED_FILE, scopes=["https://www.googleapis.com/auth/spreadsheets"])
    service = build("sheets", "v4", credentials=creds)
    res = service.spreadsheets().values().append(
        spreadsheetId=SHEETS_ID, range="GPU_Logs!A:M",
        valueInputOption="USER_ENTERED",
        body={"values": linhas}).execute()
    print(f"Enviadas {res['updates']['updatedRows']} linhas.")
